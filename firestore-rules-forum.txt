function fmEmail() { return request.auth.token.email; }
function fmRole() {
  return exists(/databases/$(database)/documents/users/$(fmEmail()))
    ? get(/databases/$(database)/documents/users/$(fmEmail())).data.get("role", "") : "";
}
function fmAdmin() { return exists(/databases/$(database)/documents/admins/$(fmEmail())); }
function fmYoung() { return fmRole() == "מרכז צעיר"; }
function fmBoger() { return fmRole() in ["מרכז בוגר", "דרג מתנדב", "מלווה ט׳", "מדריך שכבה"]; }
function fmEditor() { return fmAdmin() || fmYoung() || fmBoger(); }
function fmGearApprover() { return fmAdmin() || !(fmRole() in ["דרג מתנדב", "מלווה ט׳", "מדריך שכבה"]); }

// Calendar: written only by the sync job; changes from the site go through cal_ops.
match /cal/{doc} {
  allow read: if request.auth != null;
}
match /cal_ops/{id} {
  allow read: if request.auth != null;
  allow create: if request.auth != null && fmEditor()
    && request.resource.data.uid == fmEmail() && request.resource.data.state == "pending";
  allow update, delete: if request.auth != null && fmEditor() && resource.data.state == "pending";
}
// Area bookings
match /bookings/{id} {
  allow read: if request.auth != null;
  allow create: if request.auth != null && request.resource.data.uid == fmEmail();
  allow update: if request.auth != null && ((resource.data.uid == fmEmail() && request.resource.data.uid == fmEmail()) || fmAdmin());
  allow delete: if request.auth != null && (resource.data.get("uid", "") == fmEmail() || fmEditor());
}
// Equipment requests
match /gear/{id} {
  allow read: if request.auth != null && (resource.data.uid == fmEmail() || fmEditor() || fmEmail() in resource.data.get("approvers", [])
    || (resource.data.get("unit", "") != "" && resource.data.get("unit", "") == fmUnit()));
  allow create: if request.auth != null && request.resource.data.uid == fmEmail() && request.resource.data.status == "pending"
    && !(fmEmail() in request.resource.data.get("approvers", []));
  // Status changes: managers, מרכז צעיר, מרכז בוגר and approvers outside the בוגר roles. Other בוגרים only tick the shopping list.
  allow update: if request.auth != null && (
    ((fmEditor() || fmEmail() in resource.data.get("approvers", []))
      && (fmGearApprover() || request.resource.data.diff(resource.data).affectedKeys().hasOnly(["items"])))
    || (resource.data.uid == fmEmail() && request.resource.data.uid == fmEmail()
        && (request.resource.data.status == "pending" || request.resource.data.diff(resource.data).affectedKeys().hasOnly(["dismissed", "ownerSeen"]))));
  allow delete: if request.auth != null && (resource.data.uid == fmEmail() || fmAdmin());
}
// Useful links: everyone reads, managers write.
match /links/{id} {
  allow read: if request.auth != null;
  allow write: if request.auth != null && fmAdmin();
}
